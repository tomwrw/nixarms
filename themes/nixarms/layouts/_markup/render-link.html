{{- /* Markdown links. Internal links (starting with /) are checked at build time: a link to a page or file that doesn't exist fails the build instead of shipping a 404. */ -}} {{- $dest := .Destination -}} {{- if strings.HasPrefix $dest "/" -}} {{- $u := urls.Parse $dest -}} {{- $path := $u.Path -}} {{- if not (or (site.GetPage $path) (fileExists (path.Join "static" $path)) (fileExists (path.Join "themes/nixarms/static" $path))) -}} {{- errorf "Broken internal link %q on page %s" $dest .Page.RelPermalink -}} {{- end -}} {{- $dest = relURL $dest -}} {{- end -}} {{ .Text }} {{- /**/ -}}
